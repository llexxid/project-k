326:         private bool setHp(ulong damage)
327:         {
328:             long totalHp = _stat._hp + _stat._extraHp;
329:             //죽는경우
330:             if (totalHp - (long)damage <= 0)
331:             {
332:                 OnDead();
333:                 return false;
334:             }
335: 
336:             //ExtraHp먼저 깍기
337:             if ((long)damage > _stat._extraHp)
338:             {
339:                 long remainDamage = (long)damage - _stat._extraHp;
340:                 _stat._extraHp = 0;
341:                 _stat._hp -= remainDamage;
342:                 return true;
343:             }
344: 
345:             _stat._extraHp = _stat._extraHp - (int)damage;
346:             return true;
347:         }
