
====SkillManager====
1: using UnityEngine;
2: using System.Collections.Generic;
3: 
4: public class SkillManager : MonoBehaviour
5: {
6:     public SkillObjectPool pool;
7:     public SkillDatabase skillDatabase;
8:     float remainingTime = 0f;
9: 
10:     // 각 스킬의 다음 사용 가능 시간을 저장하는 딕셔너리
11:     private Dictionary<string, float> _skillCooldowns = new Dictionary<string, float>();
12: 
13:     public float ActivateSkill(string skillName)
14:     {
15:         SkillData data = skillDatabase.GetSkill(skillName);
16:         if (data == null)
17:         {
18:             Debug.LogWarning($"스킬 데이터가 없습니다: {skillName}");
19:             return remainingTime;
20:         }
21: 
22:         // 1. 패시브 스킬 로직 (쿨타임 미적용)
23:         if (data.skillType == SkillType.Passive)
24:         {
25:             Debug.Log($"{data.skillName} 패시브 효과 적용 중...");
26:             return remainingTime;
27:         }
28: 
29:         // 2. 쿨타임 체크
30:         if (_skillCooldowns.TryGetValue(skillName, out float nextReadyTime))
31:         {
32:             Debug.Log(remainingTime + " : 556");
33: 
34:             if (Time.time < nextReadyTime)
35:             {
36:                 Debug.Log($"{skillName} 쿨타임 중: {nextReadyTime - Time.time:F1}초 남음");
37: 
38:                 remainingTime = nextReadyTime - Time.time;
39: 
40:                 return remainingTime;
41:             }
42:             else 
43:             {
44:                 Debug.Log($"{skillName} 쿨타임 끝, 재사용");
45:                 _skillCooldowns[skillName] = Time.time + data.cooldown;
46:             }
47:         }
48:         else
49:         {
50:             Debug.Log($"{skillName} 처음 사용");
51:             _skillCooldowns[skillName] = Time.time + data.cooldown;
52:         }
53: 
54:             return remainingTime;
55:     }
56: }
====PlayerOrder====
1: using System.Collections.Generic;
2: using UnityEngine;
3: 
4: public class PlayerOrder
5: 
6: {
7:     public Node _rootNode;
8: 
9:     private PlayerDetection _detection;
10:     private PlayerMove _move;
11:     public PlayerAttack _attack;
12: 
13:     public void Init(Player player)
14:     {
15: 
16:         _detection = new PlayerDetection(player);
17:         _move = new PlayerMove(player);
18:         _attack = new PlayerAttack(player, _detection);
19: 
20:         // 트리 조립: Selector(전투 OR 대기)
21:         _rootNode = new Selector(new List<Node> // Selector노드 사용
22:         {
23:             // 1. 전투 시퀀스 (감지 -> 이동 -> 공격)
24:             new Sequence(new List<Node>
25:             {
26:                 new PlayerDetection.DetectionNode(_detection),
27:                 new PlayerMove.MoveNode(_move),
28:                 new PlayerAttack.AttackNode(_attack)
29:             }),
30:             
31:             // 2. 대기 (전투 실패 시 실행)
32:             new IdleNode()
33:         });
34:     }
35: 
36:     // 간단한 대기 노드
37:     public class IdleNode : Node
38:     {
39:         public override NodeState Evaluate()
40:         {
41:             return NodeState.Success;
42:         }
43:     }
44: }
