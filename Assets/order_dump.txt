1: using System.Collections.Generic;
2: using UnityEngine;
3: 
4: public class PlayerOrder
5: 
6: {
7:     public Node _rootNode;
8: 
9:     private PlayerDetection _detection;
10:     private PlayerMove _move;
11:     public PlayerAttack _attack;
12: 
13:     public void Init(Player player)
14:     {
15: 
16:         _detection = new PlayerDetection(player);
17:         _move = new PlayerMove(player);
18:         _attack = new PlayerAttack(player, _detection);
19:         var _idle = new PlayerIdle();
20: 
21:         // 트리 조립: Selector(전투 OR 대기)
22:         _rootNode = new Selector(new List<Node> // Selector노드 사용
23:         {
24:             // 1. 전투 시퀀스 (감지 -> 이동 -> 공격)
25:             new Sequence(new List<Node>
26:             {
27:                 new PlayerDetection.DetectionNode(_detection),
28:                 new PlayerMove.MoveNode(_move),
29:                 new PlayerAttack.AttackNode(_attack)
30:             }),
31:             
32:             // 2. 대기 (전투 실패 시 실행) — PlayerIdle.IdleNode 사용
33:             new PlayerIdle.IdleNode(_idle)
34:         });
35:     }
36: 
37:     // 간단한 대기 노드
38:     public class IdleNode : Node
39:     {
40:         public override NodeState Evaluate()
41:         {
42:             return NodeState.Success;
43:         }
44:     }
45: }
