1: using Scripts.Core;
2: using Scripts.Core.inteface;
3: using Scripts.Monster;
4: using System.Collections.Generic;
5: using UnityEngine;
6: 
7: /// <summary>
8: /// 플레이어의 공격 로직을 담당하는 클래스.
9: /// 일반 공격과 스킬 공격을 쿨타임 기반으로 관리하며,
10: /// 범위 내 적을 탐지하여 즉시 데미지를 적용한다.
11: /// </summary>
12: public class PlayerAttack
13: {
14:     // 적 탐지를 담당하는 컴포넌트 (타겟 초기화 시 참조)
15:     [SerializeField] private PlayerDetection _detection;
16: 
17:     // 공격 애니메이션 한 사이클 길이 (초). attackRate의 최솟값 기준으로도 사용
18:     private const float ANIMATION_DURATION = 0.4f;
19: 
20:     // 일반 공격 간격 (초). 이 값이 작을수록 공격 속도가 빠름
21:     // 공격 애니메이션 길이(ANIMATION_DURATION)와 맞춰 기본값 설정 → 애니메이션 1사이클 = 공격 1회
22:     public float attackRate = ANIMATION_DURATION;
23:     // 다음 공격이 가능한 시각 (Time.time 기준). 현재 시간이 이 값 이상일 때만 공격 가능
24:     private float _nextAttackTime = 0f;
25: 
26:     // 스킬 활성화 및 데이터 참조용
27:     public SkillManager skillManager;
28:     public SkillDatabase skillDatabase;
29:     public VFXManager vfxManager;
30: 
31:     // 공격이 닿는 범위 (반지름, 단위: 유니티 단위)
32:     public float attackRadius = 3f;
33:     // Physics2D.OverlapCircle 결과를 재사용하기 위한 버퍼 (GC 최소화)
34:     private List<Collider2D> _hitResults = new List<Collider2D>();
35: 
36:     // 스킬별 쿨타임 추적: 스킬 이름 → 다음 사용 가능한 Time.time
37:     private Dictionary<string, float> _skillCooldowns = new Dictionary<string, float>();
38: 
39:     // 공격 주체인 플레이어 참조
40:     [SerializeField]
41:     public Player player;
42: 
43:     // 6번 레이어(Enemy)만 탐지하도록 설정한 레이어 마스크
44:     LayerMask enemyLayer = 1 << 6;
45: 
46:     /// <summary>
47:     /// 생성자: Player로부터 필요한 참조를 가져오고 스킬 쿨타임을 초기화한다.
48:     /// </summary>
49:     public PlayerAttack(Player player, PlayerDetection detection = null)
50:     {
51:         this.player = player;
52:         this._detection = detection;
53:         this.skillDatabase = player.skillDatabase;
54:         this.skillManager = player.skillManager;
55: 
56:         // attackRate가 0 이하면 애니메이션 길이로 자동 보정
57:         // (외부에서 잘못된 값이 주입되어 매 프레임 공격되는 상황 방지)
58:         if (attackRate <= 0f)
59:         {
60:             Debug.LogWarning($"[PlayerAttack] attackRate가 {attackRate}로 설정되어 있습니다. " +
61:                              $"애니메이션 길이({ANIMATION_DURATION}초)로 자동 보정합니다.");
62:             attackRate = ANIMATION_DURATION;
63:         }
64: 
65:         // Wind_Lance 스킬을 게임 시작 즉시 사용할 수 있도록 쿨타임을 0으로 초기화
66:         _skillCooldowns["Wind_Lance"] = 0f;
67:     }
68: 
69:     /// <summary>
70:     /// 행동 트리에서 매 프레임 호출되는 공격 판정 함수.
71:     /// 쿨타임 → 범위 탐지 → 스킬/일반 공격 분기 순으로 처리한다.
72:     /// </summary>
73:     /// <returns>공격이 실행되면 Success, 쿨타임 중이거나 범위 내 적이 없으면 Failure</returns>
74:     public NodeState Attack()
75:     {
76:         // [1단계] 일반 공격 쿨타임 체크
77:         // 아직 쿨타임이 남아 있으면 이번 프레임은 공격하지 않음
78:         if (Time.time < _nextAttackTime) return NodeState.Failure;
79: 
80:         // [2단계] 적 탐지 필터 설정
81:         // enemyLayer에 속한 Trigger 콜라이더만 탐지
82:         ContactFilter2D filter = new ContactFilter2D();
83:         filter.SetLayerMask(enemyLayer);
84:         filter.useLayerMask = true;
85:         filter.useTriggers = true;
86: 
87:         // [3단계] 플레이어 중심으로 attackRadius 반경 내 적 탐지
88:         int hitCount = Physics2D.OverlapCircle(player.transform.position, attackRadius, filter, _hitResults);
89: 
90:         // 범위 내 적이 한 명도 없으면 공격할 필요 없음
91:         if (hitCount == 0) return NodeState.Failure;
92: 
93:         // [4단계] 쿨타임 소모 (적이 있을 때만 소모하여 낭비 방지)
94:         _nextAttackTime = Time.time + attackRate;
95: 
96:         // [5단계] PlayerStatus에서 기본 공격력(Atk) 조회
97:         int baseAtk = player.playerStatus?.Atk ?? 0;
98: 
99:         // [6단계] 스킬 데이터 및 쿨타임 조회
100:         string skillName = "Wind_Lance";
101:         SkillData data = skillDatabase?.GetSkill(skillName);
102:         // 해당 스킬의 다음 사용 가능 시각 (없으면 0f → 즉시 사용 가능)
103:         _skillCooldowns.TryGetValue(skillName, out float nextSkillAvailable);
104: 
105:         // [7단계] 스킬 쿨타임이 끝났으면 스킬 공격, 아니면 일반 공격 분기
106:         if (data != null && Time.time >= nextSkillAvailable)
107:         {
108:             // ══ 스킬 공격 분기 ══
109:             // 최종 데미지 = 기본 Atk × 스킬 계수(SkillData.damage)
110:             float skillCoefficient = data.damage;
111:             int skillDamage = Mathf.RoundToInt(baseAtk * skillCoefficient);
112: 
113:             Debug.Log($"[스킬 공격] Atk: {baseAtk} × 계수({data.skillName}): {skillCoefficient} = 데미지: {skillDamage}");
114: 
115:             // 스킬 쿨타임 갱신 후 스킬 매니저에 활성화 요청 (이펙트·버프 등 처리)
116:             _skillCooldowns[skillName] = Time.time + data.cooldown;
117:             skillManager?.ActivateSkill(skillName);
118: 
119:             // 범위 내 모든 적에게 스킬 데미지 적용 (광역)
120:             bool vfxPlayed = false;
121:             for (int i = 0; i < hitCount; i++)
122:             {
123:                 if (_hitResults[i].TryGetComponent<IDamageable>(out var target))
124:                 {
125:                     // VFX는 첫 번째 적에게만 한 번 재생하여 중복 방지
126:                     if (!vfxPlayed)
127:                     {
128:                         VFXManager.Instance?.GetVFX(eVFXType.Wind_Lance, target.targetPos,
129:                             player.transform.rotation, (vfx) => { vfx?.ActiveEffect(200); });
130:                         vfxPlayed = true;
131:                     }
132: 
133:                     // 데미지 적용. 적이 사망하면 루프 즉시 중단
134:                     bool isAlive = ApplyDamage(target, (ulong)skillDamage);
135:                     if (!isAlive) break;
136:                 }
137:             }
138:         }
139:         else
140:         {
141:             // ══ 일반 공격 분기 ══
142:             // 최종 데미지 = 기본 Atk × 1.0 (계수 없음, 단일 대상)
143:             float normalCoefficient = 1.0f;
144:             int normalDamage = Mathf.RoundToInt(baseAtk * normalCoefficient);
145: 
146:             Debug.Log($"[일반 공격] Atk: {baseAtk} × 계수: {normalCoefficient} = 데미지: {normalDamage}");
147: 
148:             // 범위 내 첫 번째 적 한 명에게만 데미지 적용 (단일 타깃)
149:             for (int i = 0; i < hitCount; i++)
150:             {
151:                 if (_hitResults[i].TryGetComponent<IDamageable>(out var target))
152:                 {
153:                     ApplyDamage(target, (ulong)normalDamage);
154:                     break; // 첫 번째 적 처리 후 탈출
155:                 }
156:             }
157:         }
158: 
159:         // [8단계] 데미지와 무관하게 공격 애니메이션 재생 (시각 피드백)
160:         player.PlayAttackAnimation();
161: 
162:         return NodeState.Success;
163:     }
164: 
165:     /// <summary>
166:     /// 지정한 대상에게 계산된 데미지를 적용하고 사망 여부를 처리한다.
167:     /// </summary>
168:     /// <param name="target">데미지를 받을 IDamageable 대상</param>
169:     /// <param name="damage">적용할 최종 데미지 수치</param>
170:     /// <returns>대상이 살아있으면 true, 사망했으면 false</returns>
171:     private bool ApplyDamage(IDamageable target, ulong damage)
172:     {
173:         if (target == null)
174:         {
175:             Debug.Log("대상이 null 또는 이미 파괴됨 - 건너뜀");
176:             return false;
177:         }
178: 
179:         // DamageProxy로 데미지 수치를 IAttackable 형태로 포장하여 TakeDamage에 전달
180:         bool isAlive = target.TakeDamage(new DamageProxy(damage));
181: 
182:         if (!isAlive)
183:         {
184:             Debug.Log("Monster Is Dead!! → Idle 전환");
185: 
186:             // 플레이어를 Idle 상태로 전환하여 진행 중인 공격 모션 중단
187:             player?.TurnOnAnimation(ePlayerAction.Idle);
188: 
189:             // 현재 타겟 참조를 비워 탐지 루틴이 새 타겟을 찾도록 유도
190:             if (player != null) player.currentTarget = null;
191:             if (_detection != null) _detection.currentTarget = null;
192:         }
193:         else
194:         {
195:             Debug.Log("적 데미지 적용 완료 (대상은 아직 살아있음)");
196:         }
197: 
198:         return isAlive;
199:     }
200: 
201:     /// <summary>
202:     /// ulong 데미지를 IAttackable 인터페이스로 포장하는 내부 래퍼 클래스.
203:     /// TakeDamage(IAttackable)를 호출할 때 계산된 데미지 수치만 넘기기 위해 사용.
204:     /// </summary>
205:     private class DamageProxy : IAttackable
206:     {
207:         // 전달할 데미지 수치
208:         public ulong damage { get; private set; }
209:         // 공격자 위치(사용 안 함, 인터페이스 구현 요건)
210:         public Vector3 attackerPos => Vector3.zero;
211:         // 공격 메서드(사용 안 함, 인터페이스 구현 요건)
212:         public bool Attack(IDamageable target) => false;
213: 
214:         public DamageProxy(ulong damage)
215:         {
216:             this.damage = damage;
217:         }
218:     }
219: 
220:     /// <summary>
221:     /// 행동 트리에서 PlayerAttack.Attack()을 노드로 감싸는 래퍼 클래스.
222:     /// PlayerOrder에서 이 노드를 트리에 등록하여 매 프레임 Evaluate()가 호출된다.
223:     /// </summary>
224:     public class AttackNode : Node
225:     {
226:         private PlayerAttack _attack;
227:         public AttackNode(PlayerAttack attack) { _attack = attack; }
228:         public override NodeState Evaluate() => _attack.Attack();
229:     }
230: }
